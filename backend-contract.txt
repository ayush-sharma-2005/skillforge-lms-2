import { z } from "zod";

const objectIdRegex = /^[0-9a-fA-F]{24}$/;

export const rubricItemSchema = z.object({
  criterion: z.string().min(2, "Criterion must be at least 2 characters"),
  maxPoints: z.number().min(1, "Max points must be at least 1"),
  description: z.string().optional(),
});

export const createAssignmentSchema = z.object({
  courseId: z
    .string()
    .regex(objectIdRegex, "Invalid course ID format"),
  lessonId: z
    .string()
    .regex(objectIdRegex, "Invalid lesson ID format")
    .optional(),
  title: z
    .string()
    .min(3, "Title must be at least 3 characters")
    .max(200, "Title cannot exceed 200 characters"),
  description: z
    .string()
    .min(10, "Description must be at least 10 characters"),
  assignmentType: z.enum(["text", "code"]).default("text"),
  allowedLanguages: z.array(z.string().trim().toLowerCase()).optional(),
  dueDate: z
    .string()
    .refine((val) => !isNaN(Date.parse(val)), {
      message: "Invalid date format for dueDate",
    }),
  maxScore: z.number().min(1).default(100),
  rubrics: z.array(rubricItemSchema).optional().default([]),
  similarityThreshold: z.number().min(0).max(100).default(40),
  isPeerReviewEnabled: z.boolean().default(false),
  peerReviewsPerSubmission: z.number().min(1).max(5).default(2),
  peerReviewDeadline: z
    .string()
    .refine((val) => !isNaN(Date.parse(val)), {
      message: "Invalid date format for peerReviewDeadline",
    })
    .optional(),
  aiEvaluationEnabled: z.boolean().default(true),
  isPublished: z.boolean().default(true),
});

export const updateAssignmentSchema = createAssignmentSchema.partial().omit({
  courseId: true,
});

export const submitAssignmentSchema = z.object({
  content: z
    .string()
    .min(10, "Submission content must be at least 10 characters"),
  language: z.string().trim().toLowerCase().optional(),
});

export const gradeSubmissionSchema = z.object({
  score: z.number().min(0, "Score cannot be negative").optional(),
  feedback: z.string().min(2, "Feedback must be at least 2 characters").optional(),
  rubricScores: z
    .array(
      z.object({
        criterion: z.string(),
        score: z.number().min(0),
        feedback: z.string().optional(),
      })
    )
    .optional(),
  useAiScore: z.boolean().optional(),
});

export type CreateAssignmentInput = z.infer<typeof createAssignmentSchema>;
export type UpdateAssignmentInput = z.infer<typeof updateAssignmentSchema>;
export type SubmitAssignmentInput = z.infer<typeof submitAssignmentSchema>;
export type GradeSubmissionInput = z.infer<typeof gradeSubmissionSchema>;

export interface AuthRequest extends Request {

export const authenticate = async (
  req: AuthRequest,
  _res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const authHeader = req.headers?.authorization;
    const bearerToken =
      typeof authHeader === "string" && authHeader.startsWith("Bearer ")
        ? authHeader.slice(7).trim()
        : undefined;
    const token = req.cookies?.sf_token || bearerToken;

    if (!token) {
      throw new ApiError(401, "Authentication required");
    }

    const secret = process.env.JWT_SECRET;

    if (!secret) {
      throw new Error("JWT_SECRET is not configured");
    }

    const decoded = jwt.verify(token, secret);

    if (typeof decoded !== "object" || !decoded.sub) {
      throw new ApiError(401, "Invalid authentication token");
    }

    const user = await User.findById(decoded.sub).select(
      "_id role isActive organization"
    );

    if (!user) {
      throw new ApiError(401, "User no longer exists");
    }

    // Check whether account is active
    if (!user.isActive) {
      throw new ApiError(
        403,
        "Your account has been deactivated. Please contact administrator."
      );
    }

    req.userId = user.id;
    req.userRole = user.role;
    req.organizationId = user.organization ? user.organization.toString() : undefined;

    next();
  } catch (error) {
    next(
      error instanceof ApiError
        ? error
        : new ApiError(401, "Invalid or expired authentication token")
    );
  }
};

export const enforceTenantIsolation = async (
  req: AuthRequest,
  _res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const userId = req.userId;
    const userRole = req.userRole;

    if (!userId) {
      throw new ApiError(401, "Authentication required");
    }

    // System super-admins have cross-tenant oversight
    if (userRole === "admin") {
      return next();
    }

    if (userRole !== "org_admin") {
      throw new ApiError(
        403,
        "You do not have permission to access organization management"
      );
    }

    // Resolve target organization ID from route params or body
    const rawTargetId =
      req.params.id ||
      req.params.orgId ||
      req.body.organizationId ||
      req.query.organizationId;

    const targetOrgId =
      typeof rawTargetId === "string" ? rawTargetId.trim() : undefined;

    if (targetOrgId && !mongoose.Types.ObjectId.isValid(targetOrgId)) {
      throw new ApiError(400, "Invalid organization ID format");

    if (targetOrgId && authorizedOrg._id.toString() !== targetOrgId) {

        403,

import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react'
import { useQueryClient } from '@tanstack/react-query'
import { api, setUnauthorizedHandler } from '@/lib/api'
import type { AuthResponse, User } from '@/types'

interface AuthContextValue {
  user: User | null
  /** True only while we are checking for an existing session on first load. */
  loading: boolean
  login: (input: { email: string; password: string }) => Promise<User>
  register: (input: { name: string; email: string; password: string }) => Promise<User>
  logout: () => Promise<void>
}

const AuthContext = createContext<AuthContextValue | null>(null)

export function AuthProvider({ children }: { children: ReactNode }) {
  const queryClient = useQueryClient()
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  // Restore the session from the httpOnly cookie. We use /users/profile because
  // /auth/me on the server currently returns only { userId, role }.
  useEffect(() => {
    const controller = new AbortController()
    api
      .get<AuthResponse>('/users/profile', undefined, controller.signal)
      .then((data) => setUser(data.user))
      .catch(() => setUser(null))
      .finally(() => {
        if (!controller.signal.aborted) setLoading(false)
      })
    return () => controller.abort()
  }, [])

  // If any request comes back 401 later, drop the local session.
  useEffect(() => {
    setUnauthorizedHandler(() => {
      setUser(null)
      queryClient.clear()
    })
    return () => setUnauthorizedHandler(null)
  }, [queryClient])

  const login = useCallback<AuthContextValue['login']>(async (input) => {
    const data = await api.post<AuthResponse>('/auth/login', input)
    setUser(data.user)
    return data.user
  }, [])

  const register = useCallback<AuthContextValue['register']>(async (input) => {
    const data = await api.post<AuthResponse>('/auth/register', input)
    setUser(data.user)
    return data.user
  }, [])

  const logout = useCallback<AuthContextValue['logout']>(async () => {
    try {
      await api.post('/auth/logout')
    } finally {
      setUser(null)
      queryClient.clear()
    }
  }, [queryClient])

  const value = useMemo(
    () => ({ user, loading, login, register, logout }),
    [user, loading, login, register, logout],
  )

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

// eslint-disable-next-line react-refresh/only-export-components
export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error('useAuth must be used inside AuthProvider')
  return ctx
}

import type { FieldValues, Path, UseFormSetError } from 'react-hook-form'
import { ApiError } from '@/lib/api'

/**
 * Copies server-side Zod errors onto the matching form fields.
 * Returns a banner message for anything that is not field specific.
 */
export function applyServerError<T extends FieldValues>(error: unknown, setError: UseFormSetError<T>, knownFields: readonly string[]): string | null {
  if (!(error instanceof ApiError)) return 'Something went wrong. Please try again.'

  let handled = false
  for (const item of error.fieldErrors) {
    if (knownFields.includes(item.field)) {
      setError(item.field as Path<T>, { type: 'server', message: item.message })
      handled = true
    }
  }
  return handled ? null : error.message
}

import type { Course, CourseFilters, DiscussionPost, Lesson, Paginated } from '../types';

const BASE = (import.meta as any).env?.VITE_API_URL ?? 'http://localhost:5000/api';

export class ApiError extends Error {
  status: number;
  constructor(message: string, status: number) {
    super(message);
    this.status = status;
  }
}

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = localStorage.getItem('token');
  const res = await fetch(`${BASE}${path}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...options.headers,
    },
  });
  const json = await res.json().catch(() => ({}));
  if (!res.ok) throw new ApiError(json?.message ?? 'Something went wrong. Try again.', res.status);
  return (json?.data ?? json) as T;
}

function toList<T>(raw: any, keys: string[]): T[] {
  if (Array.isArray(raw)) return raw;
  for (const k of keys) if (Array.isArray(raw?.[k])) return raw[k];
  return [];
}

export const api = {
  async listCourses(f: Partial<CourseFilters>): Promise<Paginated<Course>> {
    const q = new URLSearchParams();
    if (f.search) q.set('search', f.search);
    if (f.category) q.set('category', f.category);
    if (f.level) q.set('level', f.level);
    if (f.price) q.set('price', f.price);
    if (f.minRating) q.set('minRating', f.minRating);
    if (f.sort) q.set('sort', f.sort);
    q.set('page', String(f.page ?? 1));
    q.set('limit', '12');
    const raw: any = await request(`/courses?${q}`);
    const items = toList<Course>(raw, ['items', 'courses']);
    return {
      items,
      total: raw?.total ?? items.length,
      page: raw?.page ?? f.page ?? 1,
      pages: raw?.pages ?? raw?.totalPages ?? 1,
    };
  },

  async getCourse(id: string): Promise<Course> {
    const raw: any = await request(`/courses/${id}`);
    return raw?.course ?? raw;
  },

  async getLessons(courseId: string): Promise<Lesson[]> {
    const raw = await request(`/lessons?courseId=${courseId}`);
    return toList<Lesson>(raw, ['lessons', 'items']).sort((a, b) => a.order - b.order);

      <Logo />
      <div className="h-1 w-40 overflow-hidden rounded-full bg-border">

/** Requires a signed-in user. Pass `roles` to also restrict by role. */
export function ProtectedRoute({ roles }: { roles?: UserRole[] }) {
  const { user, loading } = useAuth()
  const location = useLocation()

  if (loading) return <FullPageLoader />
  if (!user) return <Navigate to="/login" replace state={{ from: location.pathname }} />
  if (roles && !roles.includes(user.role)) return <Navigate to="/dashboard" replace />

  return <Outlet />
}

/** Public-only wrapper that waits for session restore so the login page does not flash. */
export function SessionGate({ children }: { children: ReactNode }) {
  const { loading } = useAuth()
  if (loading) return <FullPageLoader />
  return <>{children}</>
}
